# Code

```cpp
class Solution {
public:
    int distinctAverages(vector<int>& nums) {
        set<float> avgs; 

        sort(nums.begin(), nums.end());

        int first = 0;
        int last = nums.size() - 1; 
        
        while (first < last) {
            avgs.insert(((nums[first] + nums[last]) / 2.0));
            first++;
            last--;
        }

        return avgs.size();
    }
}
```
